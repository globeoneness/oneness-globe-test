rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- helpers ----------
    function signedIn() { return request.auth != null; }
    function userDoc() { return /databases/$(database)/documents/users/$(request.auth.uid); }
    function me() { return get(userDoc()).data; }
    function active() { return signedIn() && exists(userDoc()) && me().active == true; }
    function isFull() { return active() && me().role == 'super'; }
    function isAdmin() { return active() && (me().role == 'super' || me().role == 'restricted'); }
    function can(p) { return isFull() || (active() && me().role == 'restricted' && p in me().perms); }
    function notSetUp() { return !exists(/databases/$(database)/documents/meta/setup); }
    function own() { return signedIn() && request.resource.data.userId == request.auth.uid; }
    function isOwner() { return signedIn() && resource.data.userId == request.auth.uid; }
    function onlyStatusChange() {
      return request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly(['status', 'decidedBy', 'decidedAt']);
    }

    // ---------- setup & settings ----------
    match /meta/setup { allow read: if true; allow create: if signedIn() && notSetUp(); }
    match /meta/settings {
      allow read: if active();
      allow create: if signedIn() && notSetUp();
      allow update: if isFull();
    }
    match /meta/counters {
      allow read, write: if isAdmin() || (signedIn() && notSetUp());
    }

    // ---------- staff ----------
    match /users/{uid} {
      allow read: if isAdmin() || request.auth.uid == uid;
      allow create: if
        (request.auth.uid == uid && notSetUp() && request.resource.data.role == 'super')
        || isFull()
        || (can('manage_workers') && request.resource.data.role == 'worker');
      allow update: if
        // full-access admins can never be demoted or disabled
        (resource.data.role != 'super'
           || (request.resource.data.role == 'super' && request.resource.data.active == true))
        && request.resource.data.role == resource.data.role
        && (isFull()
            || (can('manage_workers') && resource.data.role == 'worker'
                && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['perms', 'role'])));
      allow delete: if false;
    }
    match /salaries/{uid} {
      allow read: if can('view_salary') || request.auth.uid == uid;
      allow write: if can('view_salary');
    }

    // ---------- dealers & doctors ----------
    match /dealers/{id} { allow read: if active(); allow write: if can('manage_dealers'); }
    match /doctors/{id} { allow read: if active(); allow write: if can('manage_doctors'); }

    // ---------- duty records & photos (cannot be edited or deleted) ----------
    match /events/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own();
      allow update, delete: if false;
    }
    match /photos/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own();
      allow update, delete: if false;
    }
    match /lastloc/{uid} {
      allow read: if isAdmin() || request.auth.uid == uid;
      allow write: if active() && request.auth.uid == uid;
    }

    // ---------- travel fares & leave ----------
    match /expenses/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.status == 'pending';
      allow update: if can('approve_ta') && onlyStatusChange() && resource.data.userId != request.auth.uid;
      allow delete: if false;
    }
    match /leaves/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.status == 'pending';
      allow update: if can('approve_leave') && onlyStatusChange() && resource.data.userId != request.auth.uid;
      allow delete: if false;
    }

    // ---------- activity log (only full-access admins can read; nobody can edit) ----------
    match /logs/{id} {
      allow read: if isFull();
      allow create: if signedIn() && request.resource.data.by == request.auth.uid;
      allow update, delete: if false;
    }

    // ---------- products, depots, stock, promotions ----------
    match /products/{id} { allow read: if active(); allow write: if can('manage_products'); }
    match /depots/{id} { allow read: if active(); allow write: if can('manage_products'); }
    match /promotions/{id} { allow read: if active(); allow write: if can('manage_products'); }
    match /stock/{id} {
      allow read: if active();
      allow write: if can('manage_products') || can('approve_orders');
    }

    // ---------- orders (MPO submits; review -> final approval) ----------
    match /orders/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.status == 'pending';
      allow update: if request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['status', 'decidedBy', 'decidedAt', 'history'])
        && ((can('review_orders') && resource.data.status == 'pending'
              && request.resource.data.status in ['reviewed', 'rejected'])
          || (can('approve_orders') && resource.data.status in ['pending', 'reviewed']
              && request.resource.data.status in ['approved', 'rejected']));
      allow delete: if false;
    }

    // ---------- dues, collections, cash handover ----------
    match /dues/{id} {
      allow read: if active();
      allow write: if can('approve_orders') || can('approve_collections');
    }
    match /collections/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.status == 'pending';
      allow update: if can('approve_collections') && onlyStatusChange();
      allow delete: if false;
    }
    match /handovers/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.status == 'pending';
      allow update: if can('approve_collections') && onlyStatusChange();
      allow delete: if false;
    }

    // ---------- targets & tasks ----------
    match /targets/{id} {
      allow read: if isAdmin() || isOwner();
      allow write: if can('manage_targets');
    }
    match /tasks/{id} {
      allow read: if isAdmin() || isOwner();
      allow create, delete: if can('manage_tasks');
      allow update: if can('manage_tasks')
        || (isOwner() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
             ['status', 'startedAt', 'completedAt', 'report', 'startedLat', 'startedLng', 'completedLat', 'completedLng']));
    }

    // ---------- farmers ----------
    match /farmers/{id} {
      allow read, create, update: if active();
      allow delete: if false;
    }

    // ---------- account requests (approved only by Full-access Admins) ----------
    match /requests/{uid} {
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.status == 'pending'
        && request.resource.data.id == uid
        && request.resource.data.wantRole in ['staff', 'admin'];
      allow read: if isFull() || (signedIn() && request.auth.uid == uid);
      allow update: if isFull();
      allow delete: if false;
    }

    // ---------- fake-location flags ----------
    match /flags/{id} {
      allow read: if isAdmin() || isOwner();
      allow create: if active() && own() && request.resource.data.cleared == false;
      allow update: if (isFull() || can('manage_workers'))
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cleared', 'clearedBy']);
      allow delete: if false;
    }

    // ---------- one-shot location fix from the native app (for check-in) ----------
    match /livefix/{uid} {
      allow read: if isAdmin() || request.auth.uid == uid;
      allow write: if active() && request.auth.uid == uid;
    }

    // ---------- server time check ----------
    match /timesync/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }
